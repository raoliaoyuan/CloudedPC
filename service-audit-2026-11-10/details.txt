17: 
18: # 启动 envd 进程（后台运行）
19: # 使用 setsid 完全脱离控制终端，避免 envd 因 tty 问题被挂起
20: setsid $ENVD_BIN -isnotfc </dev/null &
21: ENVD_PID=$!
22: 
23: # 收集环境变量并生成包含 E2B_SANDBOX_ID 的 JSON
24: ENVVARS_JSON=$(
25:   env | grep -vE '^(SHELL|PWD|SHLVL|PATH|LS_COLORS)=' | jq -R '
26:     split("=") as $parts |
27:     {($parts[0]): ($parts[1:] | join("="))}
28:   ' | jq -s --arg sandbox_id "$X_IDE_SPACE_KEY" '
29:     add | {
30:       envVars: .,
31:       defaultWorkdir: "/workspace",
32:       E2B_SANDBOX_ID: $sandbox_id
33:     }
34:   '
35: )
36: 
37: # curl 循环重试10次，重试间隔200ms（0.2秒），直到返回2xx状态码
38: MAX_RETRIES=10
39: RETRY_DELAY=0.2  # 重试间隔改为200毫秒
40: SUCCESS=0
41: 
42: for ((i=1; i<=MAX_RETRIES; i++)); do
43:   # 发送POST请求，捕获HTTP状态码
44:   HTTP_CODE=$(
45:     curl -s -o /dev/null -w "%{http_code}" \
46:       -X POST \
47:       -H "Content-Type: application/json" \
48:       -d "$ENVVARS_JSON" \
49:       http://127.0.0.1:49983/init
50:   )
51: 
52:   # 检查状态码是否为2xx（200-299）
53:   if [[ $HTTP_CODE -ge 200 && $HTTP_CODE -lt 300 ]]; then
54:     echo "POST request succeeded (HTTP $HTTP_CODE) on attempt $i"
55:     SUCCESS=1
56:     break
57:   else
58:     echo "Attempt $i failed (HTTP $HTTP_CODE), retrying in $RETRY_DELAY second(s)..."
59:     sleep $RETRY_DELAY  # 等待200毫秒后重试
60:   fi
61: done
62: 
63: # 所有重试失败的处理
64: if [[ $SUCCESS -eq 0 ]]; then
65:   echo "Error: All $MAX_RETRIES attempts to send POST request failed"
66: fi
67: 
68: # 等待 envd 进程结束
69: wait $ENVD_PID

2: 
3: # 优先使用 shell env 中的 ENABLE_SYNC_SERVER；

11:     else
12:         echo "[sync-server] failed to get environment variables from envd, falling back to shell env"
13:     fi
14: fi
15: 
16: # 检查 sync-server 功能是否开启
17: if [ "$ENABLE_SYNC_SERVER_VALUE" != "true" ]; then
18:     echo "sync-server is disabled, skipping..."
19:     exit 0
20: fi
21: 
22: SYNC_DIR="/usr/local/tna/sync_server"
23: 
24: # 检查 sync-server 是否已安装
25: if [ ! -x "$SYNC_DIR/scripts/start.sh" ]; then
26:     echo "[sync-server] $SYNC_DIR not installed, skipping..."
27:     exit 0
28: fi
29: 
30: echo "Starting sync-server service..."
31: 
32: # 进入工作目录
33: cd "$SYNC_DIR" || {
34:     echo "[sync-server] failed to cd into $SYNC_DIR"
35:     exit 1
36: }
37: 
38: # tail 进程 PID（在 trap 中使用）
39: TAIL_PID=""
40: 
41: # 收到 supervisor 的 SIGTERM 时优雅停止：
42: # 1. 调用 stop.sh 关闭 sync_server 4 个业务进程
43: # 2. 杀掉后台 tail
44: # 3. 退出，让 supervisord 认为本次停机干净完成
45: graceful_stop() {
46:     echo "[sync-server] received stop signal, calling stop.sh..."
47:     bash "$SYNC_DIR/scripts/stop.sh" || echo "[sync-server] stop.sh returned non-zero"
48:     [ -n "$TAIL_PID" ] && kill "$TAIL_PID" 2>/dev/null
49:     exit 0
50: }
51: trap graceful_stop TERM
52: 
53: # 启动并自检
54: if ! bash "$SYNC_DIR/scripts/start.sh"; then
55:     echo "[sync-server] start.sh failed"
56:     exit 1
57: fi
58: sleep 3
59: if ! bash "$SYNC_DIR/scripts/check.sh"; then
60:     echo "[sync-server] check.sh failed"
61:     exit 1
62: fi
63: 
64: echo "sync-server service setup completed successfully"
65: 
66: # tail 日志放后台，让 ash 主进程留在前台接收 supervisord 信号
67: # 这样 trap 才能在 SIGTERM 到达时触发 graceful_stop（exec 会替换进程导致 trap 失效）
68: tail -F "$SYNC_DIR/log/sync_watchdog.log" "$SYNC_DIR/log/sync_mcd0.log" &
69: TAIL_PID=$!
70: wait "$TAIL_PID"

FILE log-forward-tee
1: #!/bin/sh
2: # log-forward-tee —— 沙箱日志转发 helper（B1）。
3: #
4: # 用途：作为管道下游读取某个 supervisord program 的 stdout+stderr，逐行做两件事：
5: #   1) 原样打回本进程 stdout（即 supervisord 的 /dev/stdout）—— 维持现有容器日志采集不变；
6: #   2) 另发一份到 spacelet 的 sandbox 日志 sink —— 由 spacelet 反查 sandboxId 注入后写自己
7: #      stdout，最终随 spacelet 容器标准输出被采集到 CLS。
8: #
9: # 用法（在 supervisord command 里用管道接到业务命令后面）：
10: #   <业务命令> 2>&1 | log-forward-tee <service>
11: # <service> 为服务名（如 api / box / cbc.d / e2b-envd / agent-init），会随每行转发给 sink，
12: # sink 注入成 "service" 字段，便于在 CLS 区分同一沙箱内不同服务的日志（sink 只能按源 IP 反查
13: # sandboxId、区分不了同沙箱多服务，故服务名必须由本 tee 侧携带）。
14: #
15: # 设计要点：
16: #   - 纯 POSIX sh，不用 bash 进程替换/数组，busybox ash 与 bash 均可运行（沙箱镜像 shell 混用）。
17: #   - 绝不阻塞/回压业务：第 2 路失败/未就绪时整体降级为“只本地采集不上报”，第 1 路照常打回 stdout。
18: #
19: #   - nc 用绝对路径解析，不依赖 PATH：
20: #     supervisord 直管的 program 继承 supervisord 自身 PATH（不含 $IDE_EDITOR_SERVER_DIR/bin），
21: #     init 脚本里的 `export PATH` 只影响 init 自身进程、不传播给兄弟 program。故 `command -v nc` 常为空、
22: #     裸 `nc` 报 not found（B1）。busybox 的 nc applet 实际在 $IDE_EDITOR_SERVER_DIR/bin/nc，
23: #     这里显式按该路径 + 常见系统路径探测。
24: #
25: #   - 转发用“一条长连接 + 后台重连循环”，而非“每行一个 nc 短连接”，也不是“单次 nc 不重连”：
26: #     · 每行 fork 一个 nc（旧实现）：日志量大时打爆进程/连接，且与 sink“一条连接反查一次源 IP、
27: #       逐行 scan”的模型不匹配。
28: #     · 单次 nc 不重连：warm 池预热的容器在预热期就起了本 tee，nc 也在预热期 fork。用户认领时容器
29: #       经 pivot_root 切根，切根期 nc 被杀 / 连接断，之后再没 nc —— 业务日志永久不转发（实测线上现象：
30: #       tee 主进程活着但无 nc 子进程）。
31: #     故这里：后台起一个 while 重连循环常驻持有 FIFO 读端，nc 每次从 FIFO 读、转发到 sink 的一条 TCP
32: #     长连接；nc 因切根 / 网络抖动 / sink 重启退出后，循环 sleep 1 自动重连，转发自愈恢复。主进程只持
33: #     FIFO 写端，逐行 append。整个 program 生命周期复用长连接、不 fork 海量 nc，又能扛过 pivot_root。
34: #
35: #   - 【关键：退避期必须持续排空 FIFO，绝不裸 sleep】—— 否则回压卡死业务进程（实测 P0 回归）。
36: #     FIFO 读端 fd4 由本进程树长期持有、永不关闭，故主进程写 fd3 永远不会收到 EPIPE，只会“成功”或
37: #     在 64KB 管道缓冲写满时【阻塞】。旧实现里 nc 掉线后裸 `sleep $backoff`（最长 30s），sleep 期间
38: #     没有任何人读 fd4 → FIFO 很快写满 → 主循环写 fd3 阻塞 → 停止 read → 上游业务进程（如 envd）写
39: #     stdout 也跟着阻塞 → 业务卡死、请求全超时。warm 认领 pivot_root 必杀 nc、必进退避窗口，日志量稍大
40: #     即命中，普遍卡死所有认领沙箱。
41: #     解法：退避不再裸 sleep，而是在退避时长内用 `timeout <backoff> cat <&4 >/dev/null` 持续把 fd4
42: #     排空丢弃。这样【任一时刻 fd4 恰有一个读者】——连通时是 nc、退避时是这个限时 cat——fd3 永不写满
43: #     阻塞，主循环永不停，业务零回压。退避期转发数据被丢弃（可接受，第 1 路 stdout 采集始终不受影响）。
44: #
45: #   - trap '' PIPE：保留。nc 与退避 cat 交接的极短瞬间理论上仍可能有 SIGPIPE，忽略之更稳妥。
46: 
47: # 转发目标：默认直接用 bridge 网关 172.24.0.1:18100（spacelet sink 的固定监听地址），不依赖注入 env。
48: # 为何不依赖 SANDBOX_LOG_SINK_ADDR env：sink 地址是基础设施常量（cloudstudio 网络网关固定
49: # 172.24.0.1、端口固定 18100），非每沙箱变化的动态值。而 warm 池预热的容器 supervisord 在预热时
50: # 就已启动，认领时注入的动态 env 走 envd 回调、进不到已在跑的 supervisord 子进程（本 tee 就是其一）。
51: # 若依赖 env，一旦注入时序变化就静默降级、日志悄悄丢。故默认硬编码网关地址，env 仅作运维可选覆盖。
52: SINK_ADDR="${SANDBOX_LOG_SINK_ADDR:-172.24.0.1:18100}"
53: SINK_HOST="${SINK_ADDR%:*}"
54: SINK_PORT="${SINK_ADDR##*:}"
55: 
56: # 服务名（第 1 个参数）。转发时给每行加前缀 \x01<service>\x01，sink 剥离前缀后注入成 service 字段。
57: # \x01(SOH) 是控制符，不会出现在正常日志文本 / JSON 里，作前缀分隔符冲突概率极低。
58: # 未传服务名时前缀为空标记（\x01\x01），sink 视 service 为空、不影响 sandboxId/msg。
59: SERVICE="$1"
60: SOH="$(printf '\001')"
61: SVC_PREFIX="${SOH}${SERVICE}${SOH}"
62: 
63: # 解析可用的可执行绝对路径（不依赖 PATH）。按优先级探测：镜像内 busybox bin → 常见系统路径 → PATH 兜底。
64: # supervisord 直管的 program 继承 supervisord 自身 PATH（不含 $IDE_EDITOR_SERVER_DIR/bin），故裸命令
65: # 常报 not found；这里显式按镜像内 busybox bin + 常见系统路径探测。
66: find_bin() {
67:   name="$1"
68:   for c in \
69:     "${IDE_EDITOR_SERVER_DIR:-/.PlnPyKFp4CRfFtgC1}/bin/$name" \
70:     "/usr/bin/$name" "/bin/$name" "/usr/local/bin/$name"; do
71:     [ -x "$c" ] && { printf '%s' "$c"; return 0; }
72:   done
73:   # 最后再试 PATH（多数场景为空，作兜底）。
74:   command -v "$name" 2>/dev/null && return 0
75:   return 1
76: }
77: 
78: NC_BIN="$(find_bin nc)"
79: # timeout/cat 用于“退避期持续排空 fd4”，避免裸 sleep 导致 FIFO 写满回压业务（见顶部说明）。
80: # 二者均为最终镜像内 busybox 1.34 applet（$IDE_EDITOR_SERVER_DIR/bin 下），正常一定解析得到。
81: TIMEOUT_BIN="$(find_bin timeout)"
82: CAT_BIN="$(find_bin cat)"
83: 
84: # nc 空窗期写 FIFO 会收到 SIGPIPE，忽略之，避免 tee 被信号杀死（见顶部说明）。
85: trap '' PIPE
86: 
87: # 第 2 路的常驻转发通道：FIFO + 后台 nc 退避重连循环。建不出 nc / FIFO 时 FWD 置空 → 只走第 1 路。
88: #
89: # 【关键 1：FIFO 必须用 O_RDWR(`<>`) 单次打开，绝不能先 `3>fifo` 再 `4<fifo`】
90: # FIFO 的 open 语义：只写打开(`3>fifo`)会【阻塞】直到有人打开读端；只读打开(`4<fifo`)会阻塞直到有人
91: # 打开写端。若在同一进程里顺序写成 `exec 3>fifo && exec 4<fifo`，第一句只写打开就永久阻塞（读端 open
92: # 在下一句、永远到不了）——整个 tee 进程卡在启动、主 `while read` 根本没开始，上游业务(如 envd)第一次
93: # 写 stdout 即被回压卡死。实测 busybox 1.34 ash 必现。
94: # 解法：`exec 3<>"$FIFO"` 以 O_RDWR 打开——FIFO 同时有了读端和写端(自己)，open【不阻塞】。主进程用
95: # `>&3` 写；后台循环的读端由 `4<&3` dup 得到。
96: #
97: # 【关键 2：抗 pivot_root】读写都用“已打开的 fd”，绝不按路径重开。
98: # warm 认领时容器做 pivot_root 换 mount namespace：预热期在 /tmp 建的 FIFO，pivot 后新根 /tmp 是另一个
99: # 文件系统、同名路径解析到【不同 inode】。若后台循环 nc 按 `<"$FIFO"` 路径重开读端，pivot 后会打开新根
100: # 那个空 FIFO、两端错位、转发永久断裂。故这里开完 fd 即 `rm` 路径，后续只用继承 fd(`>&3` / `<&4`)，
101: # 已打开的 fd 绑定 inode、不受 pivot 换路径影响。
102: #
103: # 【关键 3：O_RDWR 写端常在 → nc 读永不因业务 EOF 退出】
104: # 因 fd3 是 O_RDWR，FIFO 始终有一个写者(本进程)，后台 nc/cat 读 fd4 不会因“业务侧 stdin 关闭”而收到
105: # EOF——这正是长连接转发想要的(整个 program 生命周期复用)。收尾时显式关 fd3 并 kill 后台循环即可。
106: FWD=""
107: FIFO=""
108: # 节点级转发开关：仅当 SANDBOX_LOG_FORWARD=1 时才起 nc/FIFO 转发；默认（env 未设/非 "1"）只走第 1 路
109: # 本地采集，完全不起 nc。因沙箱镜像被多方复用——外部业务方用我们的镜像、其节点没有 spacelet 日志
110: # sink，nc 会永久连不上、周期性空转 fork（虽有下方软门闸退避、不回压业务，仍有累积开销），故默认不转发。
111: # 只有确有 sink 的节点由 spacelet-v2 --enable-sandbox-log-forward 经容器初始 env 塞 SANDBOX_LOG_FORWARD=1
112: # 显式开启。
113: if [ "${SANDBOX_LOG_FORWARD:-0}" = "1" ] && [ -n "$NC_BIN" ]; then
114:   FIFO="$(mktemp -u 2>/dev/null || printf '/tmp/log-fwd.%s' "$$")"
115:   if mkfifo "$FIFO" 2>/dev/null &&
116:      exec 3<>"$FIFO" 2>/dev/null &&  # fd3=读写端(O_RDWR，open 不阻塞)，主进程用 >&3 写
117:      exec 4<&3 2>/dev/null; then     # fd4=从 fd3 dup 的读端，供后台循环 nc/cat 用 <&4 读
118:     # 开完 fd 即删路径：后续 nc/cat 用 <&4、主进程用 >&3，均不再按路径访问，天然抗 pivot。
119:     rm -f "$FIFO" 2>/dev/null
120:     # 后台退避重连循环：nc 从继承的 fd4 读、转发到 sink；nc 退出即重连（sink 未就绪/网络抖动/pivot 杀 nc）。
121:     # 退避：nc 存活 < 2s（秒退，如连不上）时退避时长翻倍；≥ 2s（正常传输）则重置为 1s，保证 sink 就绪后
122:     # 快速连上。子 shell 继承 fd4，循环内不关它（关了读端会破坏配对）。
123:     #
124:     # 【关键：退避期用 timeout+cat 持续排空 fd4，绝不裸 sleep】nc 掉线到重连之间若无人读 fd4，FIFO 会
125:     # 写满导致主进程写 fd3 阻塞、进而卡死上游业务进程（见顶部说明）。故退避时长内改用
126:     # `timeout <backoff> cat <&4 >/dev/null` 持续读丢 fd4，让 fd4 全程有读者、fd3 永不写满阻塞。
127:     # 退避 = timeout 时长：既是"多久后重试 nc"，也是"这段时间持续排空 fd4"，二者天然合一。
128:     # 兜底：极端下 timeout/cat 解析不到时，退避回退为 1s 短 sleep（把最坏阻塞窗口从上限压到 1s）。
129:     #
130:     # 【长期失败软门闸】正常退避上限 30s（sink 短暂重启/抖动能 ~30s 内自愈）。但在"根本没有 sink"的
131:     # 旧 spacelet 节点上，nc 会永久连不上——若一直 30s 一轮 nc，几十壳×多 tee 累积仍有 fork 开销。故连续
132:     # 失败超过 fail_gate 次后，退避上限抬到 idle_max(300s)：仍持续排空 fd4、仍周期重试（不永久放弃，避免
133:     # 退回"转发断裂"），只是把重试从 30s 拉到 5min，几乎消除长期空转开销。nc 一旦连上(存活≥2s)立即复位。
134:     (
135:       backoff=1; min=1; max=30; idle_max=300; fail_gate=10; fails=0
136:       while :; do
137:         t0=$(date +%s 2>/dev/null || echo 0)
138:         "$NC_BIN" "$SINK_HOST" "$SINK_PORT" <&4 >/dev/null 2>&1
139:         t1=$(date +%s 2>/dev/null || echo 0)
140:         if [ "$((t1 - t0))" -ge 2 ]; then
141:           # nc 正常存活过（连上并传输）：复位退避与失败计数。
142:           backoff=$min; fails=0
143:         else
144:           # nc 秒退（多为连不上）：累计失败，退避翻倍。达门闸后把上限从 max 抬到 idle_max。
145:           fails=$((fails + 1))
146:           backoff=$((backoff * 2))
